def integrateSpectrum(wavelengths, fluxes, w_lower, w_upper):
    
    """
    This function performs a numerical integral of a spectrum.
    Arguments:
    wavelengths = a numpy array of wavelengths, in nanometers
    fluxes = a numpy array of fluxes, corresponding to w
    w_lower = the lower wavelength limit of the integration
    w_upper = the upper wavelength limit of the integration
    Returns:
    the integral of fluxes, with wavelength limits from w_lower to w_upper
    (this should be one floating point number)
    """
    new_wavelengths = []#creating a new list for the wavelengths between w_lower and w_upper
    new_fluxes = []#list pf equal length to new_wvalengths for teh fluxes at each point in the domain, between w_lower to w_upper
    
    for i in range(len(wavelengths)):#adding all of the  wavelengths and fluxes between w_lower and w_upper to tjhe new lists
        if w_lower <= wavelengths[i] <= w_upper:
            new_wavelengths.append(wavelengths[i])
            new_fluxes.append(fluxes[i])

    delta_wave = (w_upper-w_lower)/len(new_wavelengths) #calculating the duffeerence between succseive wavelength values
    
    trap_area = []
    for i in range(len(new_wavelengths)-1):#len(new_wavelengths)-1): #calulating the trapeziodal integral 
        trap_area.append(((new_fluxes[i]+new_fluxes[i+1])/2)*delta_wave)
        #print(f"\n \n at i = {i} \n trap_area = {trap_area[i]}")
        
    trap_area = np.array(trap_area)
    #print(f" \n {delta_wave}")
    #return delta_wave
    return np.sum(trap_area)

def estimateRGB(wavelengths, fluxes):
    
    """
    This function estimates the RGB color of an object, from its spectrum.
    It calculates the integrals the flux over wavelength ranges
    that correspond roughly to the red, green, and blue colors
    displayed on computer monitors.
    
    Arguments:
    wavelengths = a numpy array of wavelengths, in nanometers
    fluxes = a numpy array of fluxes, corresponding to w
    
    Returns:
    an RGB color, expressed as a three-element array,
    with R as the first element, G the second, B the third
    (In Python these values must span from 0.0 to 1.0, so you
    will need to renormalize your integrals by the maximum of
    the three values, to ensure this happens.)
    """

    red = integrateSpectrum(wavelengths, flux, 565, 660) #getting the red values from 565 nm to 660 nm

    green = integrateSpectrum(wavelengths, flux, 480, 570) #getting the green values from 480 nm to 570 nm

    blue = integrateSpectrum(wavelengths, flux, 400, 490) #getting the blue values from 400 nm to 490 nm


    #figuring out wich color is greatest
    if red > green and red > blue: #red is the greatest and will therefore have a value of 1 in the RGB list
        #print(f" red = {red} \n green = {green} \n blue = {blue}")
            
        R = red/red
        G = green/red
        B = blue/red
        RGB = [R,G,B]
        return RGB
    
    elif green > red and green > blue:#green is the greatest and will therefore have a value of 1 in the RGB list
        #print(f" green = {green}\n red = {red} \n blue = {blue}")
        G = green/green
        R = red/green
        B = blue/green
        RGB = [R,G,B]
        return RGB
    
    elif blue > red and blue > green:#blue is the greatest and will therefore have a value of 1 in the RGB list
        #print(f"blue = {blue} \n green = {green}\n red = {red}")
        B = blue/blue
        R = red/blue
        G = green/blue
        RGB = [R,G,B]

        return RGB

9¶

I'm noticing two clear and distinct patterns in the plots. If the peak wavlength is longer (greater wavelength in nanometers) the RGB value looks redder. Likewise, if the peak wavelength is shorter (lower wavelength in nanometers), the RGB value looks bluer.

I believe that this demonstrates the effect known as Doppler shifting. My rough understanding of doppler shifting is that an object moving away from an observer will produce longer wavelengths and this is called resdshift. An object moving towards an observer will produce shorter wavelengths, and this is called blueshift. My simulation shows that spectra with shorter peak wavelengths will appear bluer and spectra with longer peak wavelengths will appear redder.
