import thermal
import numpy as np
import matplotlib.pyplot as plt

class ThingThatGlows2:
    '''
    The class of ThingThatGlows python objects can handle calculations
    and visualizations related to physical things that emit radiation.'''

    def __init__(self, radius, temperature, name=""):
        '''
        Initialize a ThingThatGlows object.

        Parameters
        ----------
        radius : float
            The radius of the object, in meters
        temperature : float
            The temperature of the object, in K
        name (optional) : string
            The name of the object
        '''

        # store the radius and temperature as attributes
        self.radius = radius
        self.temperature = temperature
        # store the name, if given
        self.name = name
        
    def calculateArea(self):
        '''
        Calculate the surface area of the thing,
        in units of meters**2.
        '''
        return 4 * np.pi * self.radius**2
    
    def calculateLuminosity(self):
        '''
        Calculate the luminosity of the thing,
        in units of W.
        '''

        # the Stefan-Bolztmann constant
        sigma = 5.67e-8 # W/m**2/K**4
        powerPerArea = sigma * self.temperature**4
        return powerPerArea * self.calculateArea()

    def plotSpectrum(self):
        '''
        Plot the spectrum of light this thing emits.
        '''
        # create a grid of wavelengths
        wavelengths = np.logspace(2, 6, 1000)

        # this is the flux per unit area
        flux = thermal.planck(wavelengths, T=self.temperature)

        # we multiply by an area to get total luminosity
        specificluminosity = flux * self.calculateArea()

        # make the plot
        plt.plot(wavelengths, specificluminosity, label=self.name)
        if self.name:
            plt.legend(loc="best")
        plt.xscale('log'); plt.yscale('log')
        plt.xlabel('Wavelength (nm)'); plt.ylabel('Specific Luminosity (W/m)')



    #defining a new method inside the ThingThatGlows class 
    def getSpectrum(self, wavearray):
        '''
        the getSpectrum module accepts an array of wavelengths
        in nanometers and returns an array of the lumonisties
        of the thing we are looking at.

        '''

        #print(wavearray)
        #print(type(wavearray))
        
        #code from plotSpectrum() module for calcultatin the luminotisy

        wavelengths = wavearray
        
        # this is the flux per unit area
        flux = thermal.planck(wavelengths, T=self.temperature)

        # we multiply by an area to get total luminosity
        specificluminosity = flux * self.calculateArea()
        
        return  specificluminosity

# Planet inherits some of the properties of ThingThatGlows
class Planet(ThingThatGlows):
    '''A Planet is a ThingThatGlows, heated by a host star.'''
    
    # we modify the initialization, replacing the definition 
    def __init__(self, radius, orbit, star, albedo=0.3, name=""):
        '''Initalize a Planet object.
            radius is the planet radius, in m
            orbit is the orbital separation, in AU
            star is a Star object representing the host star
	    name(optional) is the name of the planet'''
        
        # ThingThatGlows depends on radius, so we must define it
        self.radius = radius
        
        # store the orbital separation
        self.separation = orbit*1.5e11
        
        # store an instance of a star object, associated 
        self.host = star
        
        # store the object name, if given
        self.name = name

        # set the temperature of the planet, 
        # based on its host star and orbital distance
        self.temperature = (0.25*(1 - albedo)
                            *(self.host.radius/self.separation)**2
                            *self.host.temperature**4)**0.25
