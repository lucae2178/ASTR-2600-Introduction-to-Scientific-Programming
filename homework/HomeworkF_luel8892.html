def planck(y,t):
    '''
    ------input-----
    
    planck function takes two inputs(y,t)
    y is the wavelength in meters and can be either a float or an integer
    t is the surface temperature in Kelvin and can be either a float or an integer

    -----output------

    the planck functio returns a float value of B(y,t) which is the intensity and has units of 
    Joules/ second / steradian / (meter^3)

    '''
    from math import e #importing the number e from the math module
    e = e

    
    B = (2*h*c**2)/((y**5)*(e**((h*c)/(y*k*t))-1)) #the planck function for intensity 

    B_sci_notation = f"{B:.2e}"

    return(B)

def Wein_law(T): # the homework doesn;t say I can't use Wein's Law

    -----input------
    
    T = int or float
    The temperature of the blackbody in Kelvin

    ----function-------

    the function calculates the peak wavelength using weins displacement law

    ----return------

    the function returns the peak wavelength in meters

    '''
 
    B = 2.898*10**(-3) #m*K 
    peak = B/T
    return(peak)
