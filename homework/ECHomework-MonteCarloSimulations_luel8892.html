def distance_center(x, y, x_center, y_center):
    '''

    This function will calculate the distance a particle is from the center
    of a distribution. 

    --------inputs-------------

    x: float, integer, numpy array

    x is the x position of a particle

    y: float, integer, numpy array

    y is the y position of a particle

    x_center: float or integer
    x_center is the x coordinate of the center of the distribution

    y_center: float or integer
    y_center is the y coordinate of the center of the distribution

    ----------function--------

    the distance_center function uses the formula:
    𝑑 = ((𝑥 − 𝑥_center)**2 + (𝑦 − 𝑦_center)**2)**1/2 

    where d is the distance from the center
    
    -----return---------------

    distance_center creates a numpy array of all of the values of d 
    This array is returned

    '''
    distance = np.sqrt((𝑥-𝑥_center)**2+(𝑦-𝑦_center)**2)
    return distance

def rand_decay( N, trueHalfLife, display):
    '''

    ---------inputs---------
    N: integer
    N is the number of atoms
    

    trueHalflife: float
    tureHalflife is the half-life of the atoms

    display:boolean
    display tells the rand_decay function if it should or should not print the output

    '''
    scale = trueHalfLife / np.log(2)
    #print(f"scale = {scale}")
    timeDecay = np.random.exponential(scale, N) # Generates a numpy array with N random numbers from an exponential distribution with a scale of the variable scale
    sortedarray = np.sort(timeDecay) #ordering the random list from least to greatest
    
    if N % 2 == 0: #if the value of N is even 
        middle =  int((len(sortedarray))/2)-1 #finding the lindex of the lower value of the middle
        middle2 = int((len(sortedarray))/2) #finding the lindex of the higher value of the middle

        average = (sortedarray[middle] + sortedarray[middle2])/2 # calculating the average value of the middle of the sorted array of decay times
        #print(f"{middle} and  {middle2}")
        #print(f" {sortedarray[middle]} amd  {sortedarray[middle2]}")
        #print(average)
        
        if display: #testing if you want to print the output
            print(f"the decay times of all of the atoms are: \n{sortedarray} \n \n the time at which half the atoms have decayed is, {average} seconds") #printing the results
            return(average)
        else:
            return(average)
    else: #if the value of N is odd

        
        
        middle = int((len(sortedarray) -1)/2) #finding the index vale of the middle
        #print(middle)
        #print(sortedarray[middle])
        
        if display: #testing if you want to print the output
            print(f"the decay times of all of the atoms are: \n{sortedarray} \n \n the time at which half the atoms have decayed is, {sortedarray[middle]} seconds") #printing the results
            return middle
        else:
            return middle
