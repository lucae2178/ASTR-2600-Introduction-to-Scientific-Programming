radius = 2
stars = 0
distance = []
star_dist = []
while radius <= Final_radius: 
    stars = ((4/3)*pi*(radius**3)*0.077) #stars/parsec^3
    stars_rounded = round(stars) #rounding the output of 
    distance.append(radius) #adding that radius distance to the list of distances
    star_dist.append(stars_rounded) #adding the rounded number of stars at this distance to the list of stars per distance
    
    radius += 2
    
x = np.array(distance) # creating an array for the x values of distances in parsecs
y = np.array(star_dist) # creating an array for the y values of stars at a certain distance

plt.plot(x,y)

plt.xlabel('x (parsecs from the sun)')

plt.ylabel('y (stars)')

plt.title('number of stars at a radius(parsecs) centered on the Sun')
