class Walker:
    
    '''
    The Walker class creates random walker objects that can move about in 1D space, 
    and records where they have been, and when they were there
    '''
    
    def __init__(self, stepsize=0.1):
        
        '''
        Initialize a random Walker object. When the walker takes steps,
        they will be drawn from a Gaussian distribution with a width of
        "stepsize".

        
        Optional Inputs:
        stepsize = average step size this walker takes, in arbitrary units

        The default step size without an input is stepsize=0.5
        '''
        #print(self, stepsize) #testing to make sure the __init__(self, stepsize=0.5) is taking the correct inputs

        
        # store the initialized step size
        self.stepsize = stepsize #defining a new attribute for self called stepszie
        # set the current position to 0.0
        self.x = 0.0
        # create a list to store where this walker has been
        # Initialize it with the starting position of 0.0
        # the index values of the list self.history records when they were there
        self.history = [self.x]

        
    def takeStep(self):

        
        '''
        
        takeStep(self) makes the walker take one random step with the 
        predefined step size
        
        '''
        # step will be a random number, with amplitude of stepsize
        step = self.stepsize*np.random.normal(0, 1) # this creates a random number between -1 and 1 under a normal distriubution
        # modify the current position
        self.x += step
        # store the step
        self.history.append(self.x)

        
    def takeWalk(self, N):
        
        '''
        
        This will have the walker take N random steps
        
        '''
        # take N steps
        for i in range(N):
            self.takeStep()

        
    def plotHistory(self):
        
        '''
        
        plotHistory(self)   plots the position on a graph with the number os steps N
        on the x-axis and the position on the y -axis
        
        '''
        # how many steps has this walker taken?
        N = len(self.history) - 1 # the first position in self.history is the starting point
        # plot a horizontal line at 0
        plt.axhline(0, color='gray')
        # plot all the positions the walker has visited
        plt.plot(self.history, alpha=0.3)
        # plot the current position
        plt.scatter(N, self.x)
        # label the axes
        plt.xlabel('N (# of steps taken)')
        plt.ylabel('1D position')
