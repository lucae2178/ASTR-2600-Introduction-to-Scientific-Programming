def generateModel(t, period):
    '''
    
    Generate a model for the x-position of the Earth,
    given some orbital period (in days).
    Parameters
    
    ----------
    
    t : array of time points
    period : float
    The orbital period of Earth, expressed in days.
    Returns
    
    -------
    
    model : np.array
    Modeled x positions for the Earth,
    evaluated at every time point.
    '''
    # convert the times into orbital phase
    phase = 2*np.pi*t/period
    # define the semimajor axis (1 AU, in meters)
    a = 1.496e+11
    # return a model for the x-position, given that period
    return a*np.cos(phase)

def calculateBadness(x_data, t, period, plot=False):
    '''
    
    Calculate the "badness-of-fit" for some period,
    as the sum of the squares of the residuals. This is
    basically the chi-squared, but just not scaled by
    any estimate of the intrinsic uncertainties.



    ------input-----------
    
    x_data : numpy array of x positions (meters)
    t : numpy array of times corresponding to x_data (days)
    
    period : float
    The orbital period of Earth, expressed in days.
    plot : bool
    If this keyword argument is set as `plot=True`,
    create a plot that compares the data with the model.
    If this keyword argument is set as `plot=False`,
    do not create any plots.
    
    
    ------Output------------
    

    badness : float
    A quantitative estimate of how bad the fit is,
    calculated as the sum of squares of the differences
    between the model and the data.
    '''
    # calculate the differences between our data and our model
    d = x_data
    
    m = generateModel(t, period)
    # calculate the badness of the fit

    
    b = np.sum((d-m)**2)
    # (this b is like chi-squared, but ignoring the errors)
    # this will generate a plot, if the plot keyword is set to True
    
    if plot:
        
        # plot the data
        plt.scatter(t, d, marker=".")
        # plot the model curve
        plt.plot(t, m, color='orange', linewidth=5, alpha=0.7)
        plt.ylabel('x (meters)')
        plt.xlabel('Time (days)')
        plt.title(f'P={period:.1f} days, $\sum(data - model)^2$ = {b:.3}')
        plt.show()
        
    # return the badness of fit, given the input period
    return b
